# Мой первый тест на Python

def сложить(a, b):
    return a + b

def вычесть(a, b):
    return a - b

# Тесты
print("Тест 1: сложение")
результат = сложить(2, 3)
if результат == 5:
    print("PASSED — 2 + 3 =", результат)
else:
    print("FAILED — ожидали 5, получили", результат)

print("\nТест 2: вычитание")
результат = вычесть(10, 4)
if результат == 6:
    print("PASSED — 10 - 4 =", результат)
else:
    print("FAILED — ожидали 6, получили", результат)

print("\nТест 3: намеренный провал")
результат = сложить(2, 2)
if результат == 5:
    print("PASSED")
else:
    print("FAILED — ожидали 5, получили", результат)
